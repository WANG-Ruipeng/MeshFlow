"""Read-only source-ID attribution, not another orientation solve or quality score."""
import collections, json, sys, time
from pathlib import Path
import numpy as np
ROOT=Path('/mnt/e/MeshFlow')
sys.path.insert(0,str(ROOT))
from r2.tools.c_restore_bounded_v1 import common as H

start=time.perf_counter(); dst=H.OUT/'review'; protected={}
def track(path,expected=None):
    p=Path(path); h=H.sha(p)
    if expected is not None: assert h==expected, str(p)
    protected[str(p)]=h
    return p
def read(path): return H.read(track(path))
def arrays(path,expected=None): return H.payload(track(path,expected))
def edges(faces):
    e=collections.defaultdict(list)
    for fi,face in enumerate(faces):
        for ci in range(3):
            a,b=int(face[ci]),int(face[(ci+1)%3])
            e[tuple(sorted((a,b)))].append(dict(face=fi,local_edge=ci,directed=[a,b]))
    return dict(e)
def same_direction(v): return v[0]['directed']==v[1]['directed']
def stats(e):
    twos=[v for v in e.values() if len(v)==2 and v[0]['face']!=v[1]['face']]
    return dict(edges=len(e),two_incident_edges=len(twos),same_direction_edges=sum(same_direction(v) for v in twos),opposed_edges=sum(not same_direction(v) for v in twos),nonmanifold_edges=sum(len(v)>2 for v in e.values()))
def permutation_is_odd(p):return sum(int(p[i])>int(p[j]) for i in range(3) for j in range(i+1,3))%2

reg=read(H.OUT/'registration.json')
inventory_path=Path(reg['records'][0]['condition_source']).parent.parent/'parent_inventory.json'
inventory=read(inventory_path); parent_meta={x['uid']:x for x in inventory['rows']}
legacy_path=ROOT/'native_t1/runs/control_mvp48_v1/data/manifest.json'
legacy=read(legacy_path); original_meta={x['uid']:x for x in legacy['objects']}
code_refs=[]
for rel,lo,hi,note in [
    ('datasets/mesh_dataset.py',215,223,'Vertex rank changes, then each face sorts all three vertex IDs; arbitrary odd permutations are possible.'),
    ('datasets/mesh_dataset.py',62,80,'Triangle soup canonical start rotates corners cyclically; this step preserves winding.'),
    ('datasets/mesh_dataset.py',247,260,'Random corner augmentation uses only three cyclic permutations; it does not reverse winding.'),
    ('r2/data_tools/user_conditions_v1/construct.py',66,84,'C is an exact full_target subset, carrying matching source vertex IDs and saved corner permutations.'),
    ('r2/data_tools/user_conditions_v1/inventory.py',60,75,'Saved original-to-retained faces and corner permutations explicitly reproduce model corner IDs.')]:
    p=track(ROOT/rel); lines=p.read_text(encoding='utf-8-sig').splitlines();code_refs.append(dict(path=str(p),sha256=protected[str(p)],line_start=lo,line_end=hi,excerpt='\n'.join(f'{i+1}: {lines[i]}' for i in range(lo-1,hi)),note=note,historical_registered_sha256=legacy['source_code_and_registration'].get(rel),matches_historical_registration=legacy['source_code_and_registration'].get(rel)==protected[str(p)] if rel in legacy['source_code_and_registration'] else None))

conditions={};condition_rows=[];source_edge_rows=[]
for r in reg['records']:
    cid=r['condition_id']
    if cid in conditions:continue
    a=arrays(r['condition_source'],r['condition_source_sha256']);pmeta=parent_meta[r['parent_id']];p=arrays(pmeta['npz'],pmeta['sha256'])
    ci=a['source_face_ids']; vid=a['C_source_vertex_ids']; original=p['original_source_faces'][a['original_source_face_ids']];perm=a['C_source_corner_permutations'];K=len(vid)
    assert np.array_equal(p['retained_to_original_face'][ci],a['original_source_face_ids'])
    assert np.array_equal(p['source_face_vertex_ids'][ci],vid)
    assert np.array_equal(np.take_along_axis(original,perm,axis=1),vid)
    assert H.same(p['full_target'].reshape(-1,3,3)[ci],a['C'])
    assert H.same(p['model_vertices'][vid],a['C'])
    ce,oe=edges(vid),edges(original);assert set(ce)==set(oe)
    source=ROOT/original_meta[r['parent_id']]['source_path'];source_state='NA_ORIGINAL_PATH_NOT_PRESENT'
    if source.is_file():
        raw=arrays(source,original_meta[r['parent_id']]['source_sha256']);assert np.array_equal(raw['faces'],p['original_source_faces']);source_state='HASH_AND_FULL_FACE_ARRAY_MATCH'
    cs,os=stats(ce),stats(oe)
    row=dict(condition_id=cid,parent_id=r['parent_id'],region_type=r['region_type'],tier=r['tier'],K=K,original_C_sourceID_two_incident_edges=os['two_incident_edges'],original_C_sourceID_same_direction_edges=os['same_direction_edges'],canonical_C_sourceID_two_incident_edges=cs['two_incident_edges'],canonical_C_sourceID_same_direction_edges=cs['same_direction_edges'],canonical_conflicts_previously_opposed=sum(len(v)==2 and same_direction(v) and not same_direction(oe[k]) for k,v in ce.items()),C_faces_with_odd_saved_corner_permutation=sum(permutation_is_odd(x) for x in perm),original_asset_verification=source_state,condition_source=r['condition_source'],condition_source_sha256=r['condition_source_sha256'],parent_prepared_path=pmeta['npz'],parent_prepared_sha256=pmeta['sha256'],original_asset_path=str(source),original_asset_expected_sha256=original_meta[r['parent_id']]['source_sha256'],source_mapping_exact=True)
    condition_rows.append(row);conditions[cid]=(a,ce,oe,row)
    for key,v in sorted(ce.items()):
        if len(v)!=2 or not same_direction(v):continue
        orig=oe[key];assert len(orig)==2
        source_edge_rows.append(dict(condition_id=cid,parent_id=r['parent_id'],source_edge=list(key),C_face_ids=[x['face'] for x in v],original_source_face_ids=[int(a['original_source_face_ids'][x['face']]) for x in v],canonical_incidence=v,original_incidence=orig,saved_corner_permutations=[perm[x['face']].tolist() for x in v],original_opposed=not same_direction(orig),cause='SAVED_CANONICAL_CORNER_REORDER'))

peroutput=[];indexed_evidence=[]
for r in reg['baselines']:
    a,ce,oe,cr=conditions[r['condition_id']]; K=len(a['C']);post=arrays(r['post_path'],r['post_sha256']);fp=post['faces'];v=post['vertices'];ids=a['C_source_vertex_ids']
    assert H.same(post['C'],a['C']) and H.same(post['output'][:K],a['C']) and H.same(v[fp[:K]],a['C'])
    case=H.OUT/'oriented'/r['method']/r['request_id'];d=read(case/'orientation_details.json');receipt=read(case/'receipt.json');assert receipt['status']=='PASS'
    indexed_C=edges(fp[:K]); indexed_all=edges(fp); source_to_index=collections.defaultdict(set);index_to_source=collections.defaultdict(set)
    for s,t in zip(ids.reshape(-1),fp[:K].reshape(-1)):
        source_to_index[int(s)].add(int(t));index_to_source[int(t)].add(int(s))
    assert all(len(x)==1 for x in source_to_index.values())
    indexed_direct=[e for e in d['edges'] if e['domain']=='C_C' and e['eligible_constraint'] and e['required_flip_XOR']==1]
    classified=collections.Counter()
    for e in indexed_direct:
        incidences=[]
        for f,j in zip(e['incident_face_ids'],e['incident_local_edges']):
            incidences.append(dict(C_face_id=f,local_edge=j,directed_source_IDs=[int(ids[f,j]),int(ids[f,(j+1)%3])]))
        d0,d1=[x['directed_source_IDs'] for x in incidences];keys=[tuple(sorted(x)) for x in (d0,d1)]
        if keys[0]!=keys[1]: label='INDEXED_ONLY_COORDINATE_MERGE'
        elif d0!=d1:label='SOURCE_INDEXED_DIRECTION_MISMATCH'
        else:label='CANONICAL_SOURCE_ID_CONFLICT_ALREADY_PRESENT'
        classified[label]+=1;indexed_evidence.append(dict(method=r['method'],request_id=r['request_id'],condition_id=r['condition_id'],indexed_edge_id=e['edge_id'],indexed_vertices=[e['vertex_u'],e['vertex_v']],source_incidence=incidences,classification=label))
    excluded=[]
    for key,inc in ce.items():
        if len(inc)!=2 or not same_direction(inc):continue
        mapped=tuple(sorted(next(iter(source_to_index[x])) for x in key));allinc=indexed_all[mapped]
        if len(allinc)>2:
            e=next(e for e in d['edges'] if (e['vertex_u'],e['vertex_v'])==mapped)
            assert not e['eligible_constraint'] and 'NONMANIFOLD_MORE_THAN_TWO_INCIDENCES' in e['exclusion_reasons']
            excluded.append(dict(source_edge=list(key),indexed_edge=list(mapped),indexed_edge_id=e['edge_id'],incidence=len(allinc),incident_face_ids=e['incident_face_ids'],free_incident_face_ids=[f for f in e['incident_face_ids'] if f>=K]))
    assert len(indexed_direct)+len(excluded)==cr['canonical_C_sourceID_same_direction_edges']
    peroutput.append(dict(method=r['method'],request_id=r['request_id'],candidate=r['candidate'],**cr,indexed_C_only_same_direction_edges=stats(indexed_C)['same_direction_edges'],whole_indexed_eligible_CC_same_direction_edges=len(indexed_direct),eligible_CC_conflicts_same_source_ID_edge=classified['CANONICAL_SOURCE_ID_CONFLICT_ALREADY_PRESENT'],eligible_CC_conflicts_coordinate_merge_only=classified['INDEXED_ONLY_COORDINATE_MERGE'],eligible_CC_source_indexed_direction_mismatches=classified['SOURCE_INDEXED_DIRECTION_MISMATCH'],C_indexed_vertex_groups_merging_distinct_source_IDs=sum(len(x)>1 for x in index_to_source.values()),canonical_CC_same_edges_excluded_due_to_additional_free_incidence=len(excluded),fixed_C_anchor_conflict_components=sum(x['fixed_C_anchor_conflict'] is True for x in d['components']),parity_cycle_conflict_components=sum(x['parity_conflict'] for x in d['components']),source_C_and_indexed_C_bytes_equal=True,details_path=str(case/'orientation_details.json'),details_sha256=protected[str(case/'orientation_details.json')],post_path=r['post_path'],post_sha256=r['post_sha256']))
    if excluded:indexed_evidence.append(dict(method=r['method'],request_id=r['request_id'],condition_id=r['condition_id'],canonical_conflicts_excluded_as_nonmanifold=excluded))

assert len(condition_rows)==6 and len(peroutput)==24
assert sum(r['canonical_C_sourceID_same_direction_edges'] for r in condition_rows)==28
assert all(r['original_C_sourceID_same_direction_edges']==0 for r in condition_rows)
assert all(r['eligible_CC_conflicts_coordinate_merge_only']==r['eligible_CC_source_indexed_direction_mismatches']==0 for r in peroutput)
assert all(H.sha(p)==h for p,h in protected.items())

notes='''Scope: six fixed DEV conditions, 24 saved D0_BASE/RBF POSTs. This is source-ID and saved-edge attribution only; no models, geometry repair, orientation solver, or quality scorers were executed.

print(json.dumps(dict(status='PASS',conditions=condition_rows,output_rows=len(peroutput),files=[str(p) for p in files],seconds=time.perf_counter()-start),indent=2))
