    known=set(map(int,face_ids));n=p['N'];k=len(known);f=n-k

    rows={g:[] for g in ('CC','CF','FF')}
    for contact in p['pair_contacts']:

        rows[g].append(contact)
    expected={'CC':k*(k-1)//2,'CF':k*f,'FF':f*(f-1)//2}
