"""One archived A_continue replay comparison; no model or training is invoked here."""
from pathlib import Path
import hashlib
import json
import difflib
import numpy as np

ROOT = Path("/mnt/e/MeshFlow")
AUDIT = ROOT / "native_t1/validation/a_continue_20260930_104000"
def sha(path):
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(8*1024*1024), b""):
            h.update(chunk)
    return h.hexdigest()
def read(path):
    return json.loads(path.read_text(encoding="utf-8-sig"))
def bits(a,b):
    return a.dtype == b.dtype and a.shape == b.shape and np.array_equal(a.view(np.uint32), b.view(np.uint32))

run = read(AUDIT / "replay/run.json")
assert run["status"] == "PASS"
assert run["profile"] == "a-continue"
assert run["load"]["cumulative_updates"] == 1500
assert run["state_after"] == run["load"]["state_sha256"]
assert run["counts"]["model_forward_attempts"] == run["counts"]["model_forward_returns"] == 50
assert run["counts"]["complete_rollouts"] == 1
assert run["counts"]["optimizer_updates"] == run["counts"]["model_backward_calls"] == 0
reference = ROOT / "experiments/mf_h1_local0/outputs/a_then_fc_20260930_001700/raw/A_continue_P0_K12_seed9601.npz"
assert sha(reference) == "94834c2136410a6c6fcf3cc63a5c5dfb3954967d1b6a642729605a667bf1df6d"
inputs = ROOT / "experiments/mf_h1_local0/outputs/native_patch_20260929_010000/inputs.npz"
assert sha(inputs) == "e80d52e0906bc762475f35a2526cb99ff5b52858c9d2e6a38d09018e5d5aea73"
with np.load(reference, allow_pickle=False) as old, np.load(AUDIT / "replay/raw.npz", allow_pickle=False) as new, np.load(inputs, allow_pickle=False) as source:
    old_key = "output" if "output" in old else "raw"
    trajectory_equal = bits(new["path"], old["path"])
    endpoint_equal = bits(new["output"], old[old_key])
    C = source["parent0_K12_constraints"].reshape(12,9)
    known_equal = bits(new["path"][:,:12], np.broadcast_to(C, (51,12,9)))
    free_noise_equal = bits(new["initial_gaussian"][0,12:], old["path"][0,12:])
    comparison = dict(reference=str(reference), reference_sha256=sha(reference),
                      reference_keys=old.files, actual_keys=new.files,
                      all51_states_bitwise=trajectory_equal, endpoint_bitwise=endpoint_equal,
                      known_all51_bitwise=known_equal, same_initial_free_gaussian_bitwise=free_noise_equal,
                      output_shape=list(new["output"].shape), path_shape=list(new["path"].shape))
assert all([trajectory_equal, endpoint_equal, known_equal, free_noise_equal]), comparison
before = read(AUDIT / "before_manifest.json")
intended_changed = {".gitignore", "native_t1/README.md", "native_t1/DEPENDENCIES.md", "native_t1/__main__.py"}
protection, diff = [], []
for item in before:
    relative = item["path"].replace("\\","/")
    current = ROOT / relative
    actual = sha(current)
    changed = actual != item["sha256"]
    if relative not in intended_changed:
        assert not changed, relative
    protection.append(dict(path=relative, before_sha256=item["sha256"], after_sha256=actual, changed=changed))
    if changed:
        backup = AUDIT / "before" / relative
        if backup.suffix == ".py":
            backup = backup.with_suffix(".py.txt")
        diff.extend(difflib.unified_diff(backup.read_text(encoding="utf-8-sig").splitlines(keepends=True),
                                       current.read_text(encoding="utf-8-sig").splitlines(keepends=True),
                                       fromfile="a/"+relative, tofile="b/"+relative))
for relative in ["WORKSPACE.md", "native_t1/baseline.py", "native_t1/tests/test_baseline.py"]:
    current = ROOT / relative
    diff.extend(difflib.unified_diff([], current.read_text(encoding="utf-8-sig").splitlines(keepends=True),
                                   fromfile="/dev/null", tofile="b/"+relative))
(AUDIT / "code.diff").write_text("".join(diff), encoding="utf-8")
official = read(AUDIT / "official_files.json")
for row in official:
    assert sha(ROOT / row["path"].replace("\\","/")) == row["sha256"]
report = dict(status="PASS", baseline="A_continue", cumulative_updates=1500,
              checkpoint=run["load"], runtime=run["sampling"]["runtime"], counts=run["counts"],
              comparison=comparison, package_protection=protection, official_protection=official,
              cleanup=read(AUDIT / "cleanup.json"), diff_sha256=sha(AUDIT / "code.diff"),
              tests=read(AUDIT / "tests.json"), model_sampling_seconds=run["sampling"]["seconds"],
              unsupported_or_retried_model_calls=0, new_training_updates=0,
              tooling_notes=[
                "A PowerShell/bash inline Python version query had a quoting error; direct wsl --cd command succeeded.",
                "apply_patch failed with Windows sandbox logon1385 before applying changes; authorized exec file writes used instead.",
                "Historical source/weights/data/output directories were not moved or edited; only listed Python caches removed.",
                "before/*.py.txt are small source snapshots, deliberately not importable modules."
              ])
(AUDIT / "results.json").write_text(json.dumps(report, indent=2, allow_nan=False), encoding="utf-8")
print(json.dumps(dict(status=report["status"], comparison=comparison, counts=report["counts"],
                      cleanup_bytes=report["cleanup"]["deleted_bytes"], diff=report["diff_sha256"]), indent=2))
